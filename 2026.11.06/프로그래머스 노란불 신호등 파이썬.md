def solution(signals):

#   주기 확인
    cycle = []
    for signal in signals:
        cycle.append(sum(signal))

#   최소공배수
    max_num = max(cycle)

    while True:
        flag = True
        for c in cycle:
            if max_num % c:
                flag = False
                break
        if flag == True:
            break
        else:
            max_num += 1

#   dp를 활용하여 최소 주기 확인
    dp = [0] * (max_num + 1)
    for t in range(len(cycle)):
        for i in range(1, max_num + 2):
            if t < len(cycle) - 1:
                if signals[t][0] < i % cycle[t] <= signals[t][0] + signals[t][1]:
                    dp[i] += 1
            else:
                if signals[t][0] < i % cycle[t] <= signals[t][0] + signals[t][1]:
                    dp[i] += 1
                    if dp[i] == len(signals):
                        return i
    return -1
