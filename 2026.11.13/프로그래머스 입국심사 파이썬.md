def solution(n, times):
    answer = 0
    left, right = 1, max(times) * n
    
    while left <= right:
        mid = (left + right) // 2
        total_n = 0
        for i in range(len(times)):
            total_n += mid // times[i]
        if total_n >= n:
            right = mid - 1
            answer = mid
        else:
            left = mid + 1
    
    
    return answer
