from collections import deque
from itertools import product

def solution(n, infection, edges, k):
    answer = 0
    que = deque()
    product_infection = product([1, 2, 3], repeat = k)
    adj = [[] for _ in range(n + 1)]
    for e in edges:
        adj[e[0]].append([e[1], e[2]])
        adj[e[1]].append([e[0], e[2]])
    
    for pp in product_infection:
        visited = [0] * (n + 1)
        visited[infection] = 1
        for p in pp:
            for i in range(1, n + 1):
                if visited[i] == 1:
                    que.append(i)
            while que:
                q = que.popleft()
                for a in adj[q]:
                    if visited[a[0]] == 0 and a[1] == p:
                        visited[a[0]] = 1
                        que.append(a[0])
        answer = max(answer, sum(visited))
    return answer
